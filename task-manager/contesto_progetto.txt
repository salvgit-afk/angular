@Pipe({
  name: 'troncaTesto',

})
export class TroncaTestoPipe implements PipeTransform {
  transform(valore: string, limite: number = 20): string {
    if (!valore) return '';
    return valore.length > limite ? valore.substring(0, limite) + '...' : valore;
  }

import { Component } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { TaskItemComponent } from '../task-item/task-item.component';
import { TroncaTestoPipe } from '../../tronca-testo.pipe';

interface Task {
  id: string;
  titolo: string;
  completato: boolean;
  dataCreazione: Date;
}

@Component({
  selector: 'app-tasks-list',
  standalone: true,
  imports: [CommonModule, FormsModule, TaskItemComponent, TroncaTestoPipe],
  templateUrl: './tasks-list.component.html',
  styleUrl: './tasks-list.component.css'
})
export class TasksListComponent {
  tasks: Task[] = [
    { id: crypto.randomUUID(), titolo: 'Task di prova Lezione 4', completato: false, dataCreazione: new Date() }
  ];
  nuovoTask = '';
  soloCompletati = false;

  aggiungi() {
    if (this.nuovoTask.trim()) {
      this.tasks.push({ 
        id: crypto.randomUUID(),
        titolo: this.nuovoTask, 
        completato: false, 
        dataCreazione: new Date() 
      });
      this.nuovoTask = '';
    }
  }

  elimina(index: number) {
    this.tasks.splice(index, 1);
  }

  gestisciToggle(index: number, nuovoStato: boolean) {
    this.tasks[index].completato = nuovoStato;
  }
}

import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { TroncaTestoPipe } from '../../tronca-testo.pipe';



interface Task {
  id: string;
  titolo: string;
  completato: boolean;
  dataCreazione: Date;
}

@Component({
  selector: 'app-task-item',
  standalone: true,
  imports: [CommonModule, FormsModule, TroncaTestoPipe],
  templateUrl: './task-item.component.html',
  styleUrl: './task-item.component.css'
})
export class TaskItemComponent {
  @Input({ required: true }) task!: Task;
  @Output() delete = new EventEmitter<void>();
  @Output() toggle = new EventEmitter<boolean>();

  onToggleChange() {
    this.toggle.emit(this.task.completato);
  }

  onDeleteClick() {
    this.delete.emit();
  }
